# 1. 등록 수의 추이와 경향 보기
## 월별 등록 수 추이 
```sql
with
monthly_agg as (
select
	to_char(register_date,'YYYY-MM') as year_month,
	count(distinct user_id) as register_count
from mst_users
group by 1)

select
year_month,
register_count,
lag(register_count) over (order by year_month) as last_month_count,
round(100.0*register_count / nullif(lag(register_count) over (order by year_month),0),2) as month_over_month_ratio
from monthly_agg
order by year_month
```

## 등록 디바이스별 추이

with
mst_users_with_year_month as (
select
	date_trunc('month',register_date)::date as year_month,
	user_id,
	register_device
from mst_users)

select
to_char(year_month,'YYYY-MM') as year_month,
count(user_id) as register_count,
COUNT(user_id) FILTER (WHERE register_device = 'PC_Web') AS register_pc,
COUNT(user_id) FILTER (WHERE register_device = 'Mobile_Android') AS register_android,
COUNT(user_id) FILTER (WHERE register_device = 'Mobile_iOS') AS register_ios
from mst_users_with_year_month

# 2.지속률과 정착률 산출하기
## 1~7일 정착률 계산하기 (실무용)
```sql
WITH target_users AS (
  SELECT
    user_id,
    register_date::date AS register_date,
    -- [안전장치] 데이터 분석 기준점을 '어제'로 고정
    (CURRENT_DATE - INTERVAL '1 day')::date AS yesterday_date
  FROM mst_users
  -- 오늘 가입한 유저는 어차피 어제 기준 데이터에 포함되지 않으므로 제외
  WHERE register_date::date < CURRENT_DATE
),

user_daily_activity AS (
  SELECT
    u.user_id,
    u.register_date,
    u.yesterday_date,
    -- 가입일과 액션일의 날짜 차이 계산 (Day 1, Day 2 ... Day 7)
    (a.stamp::date - u.register_date) AS days_elapsed
  FROM target_users u
  INNER JOIN action_log a -- 가입 이후 액션이 있는 기록만 매칭
    ON u.user_id = a.user_id
   AND a.stamp::date > u.register_date
   -- [성능 최적화 1] 오늘 날짜의 액션 로그는 원천 차단 (어제 데이터까지만 스캔)
   AND a.stamp::date < CURRENT_DATE
   -- [성능 최적화 2] 7일차 최대 범위인 7일치 로그만 스캔
   AND a.stamp::date <= u.register_date + INTERVAL '7 days' 
  GROUP BY u.user_id, u.register_date, u.yesterday_date, (a.stamp::date - u.register_date)
)

SELECT
  register_date,
  -- 해당 일자의 총 가입자 수
  COUNT(DISTINCT user_id) AS total_registrations,

  -- Day 1 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '1 day' <= yesterday_date AND days_elapsed = 1 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '1 day' <= yesterday_date THEN user_id END), 0), 2) AS day_1_rate,

  -- Day 2 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '2 day' <= yesterday_date AND days_elapsed = 2 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '2 day' <= yesterday_date THEN user_id END), 0), 2) AS day_2_rate,

  -- Day 3 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '3 day' <= yesterday_date AND days_elapsed = 3 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '3 day' <= yesterday_date THEN user_id END), 0), 2) AS day_3_rate,

  -- Day 4 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '4 day' <= yesterday_date AND days_elapsed = 4 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '4 day' <= yesterday_date THEN user_id END), 0), 2) AS day_4_rate,

  -- Day 5 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '5 day' <= yesterday_date AND days_elapsed = 5 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '5 day' <= yesterday_date THEN user_id END), 0), 2) AS day_5_rate,

  -- Day 6 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '6 day' <= yesterday_date AND days_elapsed = 6 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '6 day' <= yesterday_date THEN user_id END), 0), 2) AS day_6_rate,

  -- Day 7 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '7 day' <= yesterday_date AND days_elapsed = 7 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '7 day' <= yesterday_date THEN user_id END), 0), 2) AS day_7_rate

FROM user_daily_activity
GROUP BY register_date
ORDER BY register_date;
```
* **Interval 7 days** 최적화
  * 유저당 최대 가입 후 7일 동안의 로그만 탐색하므로, 테이블이 아무리 커도 조인 연산량이 대폭 제한된다.
 
* **시간적 기회 왜곡 차단**
  * 오늘이 6월 5일이라면 6월 4일 가입자의 Day 1은 6월 5일이다. 오늘이 아직 다 지나지 않았기 때문에 이 쿼리에서는 **거짓**이 되어 분모에서 안전하게 제외된다. --> 완벽하게 어제 하루를 통째로 보낸 유저들만 비교 대상이 되므로 데이터가 매우 정확해진다.

## Week 1 ~ Week 4 정착률 통합 쿼리 
```sql
WITH target_users AS (
  SELECT
    user_id,
    register_date::date AS register_date,
    -- [안전장치] 데이터 분석 기준점을 '오늘(CURRENT_DATE)'이 아닌 '어제'로 고정
    (CURRENT_DATE - INTERVAL '1 day')::date AS yesterday_date
  FROM mst_users
  -- 오늘 가입한 유저는 어차피 어제 기준 데이터에 포함되지 않으므로 제외
  WHERE register_date::date < CURRENT_DATE
),

user_weekly_activity AS (
  SELECT
    u.user_id,
    u.register_date,
    u.yesterday_date,
    -- 가입일과 액션일의 날짜 차이 계산
    (a.stamp::date - u.register_date) AS days_elapsed
  FROM target_users u
  INNER JOIN action_log a 
    ON u.user_id = a.user_id
   AND a.stamp::date > u.register_date
   -- [성능 최적화 1] 오늘 날짜의 액션 로그는 원천 차단 (어제 데이터까지만 스캔)
   AND a.stamp::date < CURRENT_DATE
   -- [성능 최적화 2] 4주차 최대 범위인 28일치 로그만 스캔
   AND a.stamp::date <= u.register_date + INTERVAL '28 days' 
  GROUP BY u.user_id, u.register_date, u.yesterday_date, (a.stamp::date - u.register_date)
)

SELECT
  register_date,
  -- 해당 일자의 총 가입자 수
  COUNT(DISTINCT user_id) AS total_registrations,

  -- Week 1 잔존율 (1~7일 사이 방문)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '7 days' <= yesterday_date AND days_elapsed BETWEEN 1 AND 7 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '7 days' <= yesterday_date THEN user_id END), 0), 2) AS retention_week_1,

  -- Week 2 잔존율 (8~14일 사이 방문)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '14 days' <= yesterday_date AND days_elapsed BETWEEN 8 AND 14 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '14 days' <= yesterday_date THEN user_id END), 0), 2) AS retention_week_2,

  -- Week 3 잔존율 (15~21일 사이 방문)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '21 days' <= yesterday_date AND days_elapsed BETWEEN 15 AND 21 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '21 days' <= yesterday_date THEN user_id END), 0), 2) AS retention_week_3,

  -- Week 4 잔존율 (22~28일 사이 방문)
  ROUND(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '28 days' <= yesterday_date AND days_elapsed BETWEEN 22 AND 28 THEN user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN register_date + INTERVAL '28 days' <= yesterday_date THEN user_id END), 0), 2) AS retention_week_4

FROM user_weekly_activity
GROUP BY register_date
ORDER BY register_date;
```

* **latest_date** --> **yesterday_date**로 변경 및 로직 단순화
  
  * 비용의 낭비 없이 **어제 날짜**를 기준점으로 삼는다.
    
* `a.stamp::date` < `current_date` 조건 추가
  
  * 오늘 쌓이고 있는 실시간 로그를 아예 읽지 않는다. --> 쿼리 스캔 비용을 획기적으로 줄인다.
    
* 완벽해진 정밀도
  
  * 어제 날짜를 기준으로 **register_date + N days <= yesterday_date** 만 분모에 들어가기 때문에, 5월 15일 가입자(Week3 종료일이 6월 5일 오늘인 유저)는 오늘이 다 지나지 않았으므로 안전하게 분모에서 제외된다. --> 이 유저는 내일이 되면 완벽한 데이터로 분모에 포함된다.

# 3. Aha! Moment /기능 채택과 리텐션의 상관관계
```sql
WITH target_users AS (
  SELECT
    user_id,
    register_date::date AS register_date,
    -- [안전장치] 데이터 분석 기준점을 '어제'로 고정
    (CURRENT_DATE - INTERVAL '1 day')::date AS yesterday_date
  FROM mst_users
  -- [정밀 컷오프] 최소한 가입 익일(Day 1)이 어제 날짜보다 작거나 같은(이미 완벽히 지난) 유저만 자격 부여
  WHERE register_date::date + INTERVAL '1 day' <= (CURRENT_DATE - INTERVAL '1 day')::date
),

mst_actions AS (
  -- 분석하고자 하는 가입 첫날의 핵심 액션 정의
  SELECT 'view' AS action
  UNION ALL SELECT 'comment' AS action
  UNION ALL SELECT 'follow' AS action
),

-- 1. 유저별 가입 당일(Day 0) 특정 액션 수행 여부 집계 (있으면 1)
user_day0_activity AS (
  SELECT
    u.user_id,
    a.action,
    1 AS do_action
  FROM target_users u
  INNER JOIN action_log a
    ON u.user_id = a.user_id
   AND a.stamp::date = u.register_date -- [성능 최적화] 가입 당일 로그만 콕 집어 스캔
  WHERE a.action IN ('view', 'comment', 'follow')
  GROUP BY u.user_id, a.action
),

-- 2. 유저별 가입 익일(Day 1) 재방문 여부 집계 (방문했으면 1)
user_day1_activity AS (
  SELECT
    u.user_id,
    1 AS index_date_action
  FROM target_users u
  INNER JOIN action_log a
    ON u.user_id = a.user_id
   AND a.stamp::date = u.register_date + INTERVAL '1 day' -- [성능 최적화] 가입 익일 로그만 콕 집어 스캔
  GROUP BY u.user_id
)

-- 3. 최종 결합 및 액션별 리텐션 비교 집계
SELECT
  m.action,
  COUNT(*) AS users, -- 분석 대상이 되는 총 유저 수
  
  -- 가입 당일 해당 액션 사용률 (%)
  ROUND(AVG(COALESCE(d0.do_action, 0)) * 100.0, 2) AS usage_rate,
  
  '01 day repeat' AS index_name,
  
  -- 가입 당일 액션을 '한' 유저들의 Day 1 잔존율 (%)
  ROUND(AVG(CASE WHEN COALESCE(d0.do_action, 0) = 1 THEN COALESCE(d1.index_date_action, 0) END) * 100.0, 2) AS idx_rate,
  
  -- 가입 당일 액션을 '안 한' 유저들의 Day 1 잔존율 (%)
  ROUND(AVG(CASE WHEN COALESCE(d0.do_action, 0) = 0 THEN COALESCE(d1.index_date_action, 0) END) * 100.0, 2) AS no_action_idx_rate

FROM target_users u
CROSS JOIN mst_actions m -- 대상 유저와 마스터 액션을 매칭 (모든 유저에게 3개 액션 기본 틀 부여)
LEFT JOIN user_day0_activity d0
  ON u.user_id = d0.user_id
 AND m.action = d0.action
LEFT JOIN user_day1_activity d1
  ON u.user_id = d1.user_id
GROUP BY m.action
ORDER BY m.action;
```
* `WHERE register_date::date + INTERVAL '1 day' <= (CURRENT_DATE - INTERVAL '1 day')::date`
  
  * 리텐션 지표가 왜곡되는 것을 막기 위한 **시간적 자격 요건(정밀 컷오프)'** 안정장치: "최소한 가입한 지 꼬박 하루(Day1)가 지나 데이터가 온전하게 다 쌓인 유저들만 분석 분모에 넣겠다."
