# 1. 사용자의 액션 수 집계하기
```sql
  WITH base AS (
      SELECT
          action,
          session, -- user_id 대신 session을 식별자로 사용
          COUNT(*) AS action_cnt
      FROM action_log
      GROUP BY action, session
  )
  SELECT
      action,
      -- 1. 해당 액션이 발생한 총 세션 수 (비회원 방문도 각각 카운트됨)
      COUNT(DISTINCT session) AS session_count, 
      
      -- 2. 세션당 해당 액션 평균 발생 횟수
      ROUND(AVG(action_cnt), 2) AS avg_count_per_session, 
      
      -- 3. 전체 세션 대비 해당 액션 발생 세션 비율 (Usage Rate)
      ROUND(100.0 * COUNT(DISTINCT session) / (SELECT COUNT(DISTINCT session) FROM action_log), 2) AS session_usage_rate
  FROM base
  GROUP BY action
  ORDER BY session_usage_rate DESC;
```
* 세션당 체류 액션 수 (Deep-dive)
  ```sql
    SELECT
      CASE WHEN user_id IS NULL THEN 'Guest' ELSE 'Member' END AS user_type,
      COUNT(DISTINCT session) AS total_sessions,
      COUNT(*) AS total_actions,
      -- 세션당 평균 액션 수 (방문 당 활동 밀도)
      ROUND(COUNT(*)::numeric / COUNT(DISTINCT session), 2) AS actions_per_session
  FROM action_log
  GROUP BY user_type;
  ```

 ## 회원과 비회원을 구분해서 집계하기
 * 회원&비회원 집계
   ```sql
    with
    session_user_map as (
   --- 각 세션별로 "로그인한 유저가 있는지" 미리 정의
     select
    	session,
    	max(user_id) as fixed_user_id,
    	min(stamp) filter (where user_id is not null) as login_time
    from action_log
    group by session)
    
    select
      l.session,
      l.user_id,
      l.action,
      l.stamp,
      case
      when m.fixed_user_id is null then 'none' -- 세션 내내 로그인 안 함
      when l.stamp>=m.login_time then 'member' -- 로그인 시점 이후
      else 'guest_before_login' end as member_status --로그인 하기 전의 활
    from action_log as l
    left join session_user_map as m
    on l.session=m.session
   ```
* 로그인 전환 유도 목적 (Conversion Focus)
  ```sql
  with
  session_metrics as (
  select
  	session,
  	min(stamp) filter (where user_id is not null) as login_timestamp
  from action_log
  group by session)
  
  select
  l.session,
  l.action,
  l.products,
  case
  when m.login_timestamp is null then 'Pure_Guest' -- 끝까지 로그인 안 함
  when l.stamp<m.login_timestamp then 'Guest_Before_Login' --로그인 전 탐색 중
  when l.stamp=m.login_timestamp then 'Login_Moment' -- 로그인 직후 액션
  else 'Member_After_Login' end as behavioral_stage, --로그인 후 활동
  l.stamp
  from action_log as l
  left join session_metrics as m
  on l.session=m.session
  ```
* 순수 매출 집계
  ```sql
    with
  session_summary as (
  select
  	session,
  	max(user_id) as final_user_id,
  	sum(amount) as session_amount,
  	count(*) filter(where action='purchase') as purchase_count
  from action_log
  group by session)
  
  select
  case
  when final_user_id is not null then 'member_session'
  else 'guest_session' end as session_type,
  count(distinct session) as total_sessions,
  sum(session_amount) as total_revenue,
  round(sum(purchase_count)::numeric/nullif(count(distinct session),0),3)
  as cvr
  from session_summary
  group by session_type
  ```
  * CVR (Conversion Rate): `구매 횟수/총 세션 수`를 계산하여, 어떤 그룹이 더 효율적으로 구매에 도달하는지 보여준다.

  ```sql
     WITH action_day_count_per_user AS (
      SELECT
          -- user_id가 없으면 guest_id를 사용해 비회원도 각각 고유하게 식별
          COALESCE(user_id, guest_id) AS distinct_user_id,
  ```

# 5.Decile 분석
* 비회원 데이터는 과감히 제외
  ```sql
    with
  user_purchase_amount as (
  select
  	user_id,
  	coalesce(sum(amount) filter(where action='purchase'),0) as purchase_amount
  from action_log
  where user_id is not null
  group by user_id),
  
  users_with_decile as (
  select
  	user_id,
  	purchase_amount,
  	case
  	when purchase_amount=0 then 10
  	else ceil(cume_dist() over (order by purchase_amount desc)*10) end as decile
  from user_purchase_amount),
  
  decile_with_purchase_amount as (
  select
  	decile,
    count(user_id) as user_count
  	sum(purchase_amount) as amount,
  	avg(purchase_amount) as avg_amount
  from users_with_decile
  group by decile)
  
  select
  decile,
  user_count,
  amount,
  avg_amount,
  100.0*amount/nullif(sum(amount) over(),0) as total_ratio,
  100.0*sum(amount) over (order by decile)/nullif(sum(amount) over (),0) as cumulative_ratio
  from decile_with_purchase_amount
  order by decile
  ```
* 가상 ID(Alternative ID)를 발급하여 추척(cookie_id,session_id)
  ```sql
    with
    user_purchase_amount as (
    select
    	coalesce(user_id, 'Guest_'||session) as blended_user_id,
    	coalesce(sum(amount) filter(where action='purchase'), 0) as purchase_amount
    from action_log
    group by blended_user_id),
    
    users_with_decile as (
    select
    	blended_user_id,
    	purchase_amount,
    	case
    		when purchase_amount = 0 then 10
    		-- 동점자 처리가 고민된다면 ntile을 활용하되, 0원 유저는 위에서 걸러지므로 안전합니다.
    		else ntile(10) over (order by purchase_amount desc) 
    	end as decile
    from user_purchase_amount),
    
    decile_with_purchase_amount as (
    select
    	decile,
    	count(blended_user_id) as user_count, -- 분위별 유저 수 추가 (실무 요건)
    	sum(purchase_amount) as amount,
    	avg(purchase_amount) as avg_amount
    from users_with_decile
    group by decile)
    
    select
    	decile,
    	user_count,
    	amount,
    	avg_amount,
    	100.0 * amount / nullif(sum(amount) over(), 0) as total_ratio,
    	100.0 * sum(amount) over (order by decile) / nullif(sum(amount) over (), 0) as cumulative_ratio
    from decile_with_purchase_amount
    order by decile;
  ```
# 6.RFM 분석
* Recency&Frequency&Monetary 지표 집계
  ```sql
      with
    base_purchase_data as (
    select
    	user_id,
    	stamp::date as purchase_date,
    	amount
    from action_log
    where user_id is not null
    and action='purchase'),
    
    max_date_cte as (
    select
    	max(purchase_date) as max_global_date
    from base_purchase_data)
    
    select
    b.user_id,
    (select max_global_date from max_date_cte)-max(b.purchase_date) as recency,
    count(*) as frequency,
    coalesce(sum(b.amount),0) as monetary
    from base_purchase_data as b
    group by b.user_id
  ```
* RFM 랭크 정의
  ```sql
      WITH base_purchase_data AS (
        SELECT
            user_id,
            stamp::date AS purchase_date,
            amount
        FROM action_log
        WHERE user_id IS NOT NULL
          AND action = 'purchase'  -- 오직 구매 유저만 대상
    ),
    
    max_date_cte AS (
        SELECT MAX(purchase_date) AS max_global_date FROM base_purchase_data
    ),
    
    user_rfm_raw AS (
        SELECT
            b.user_id,
            (SELECT max_global_date FROM max_date_cte) - MAX(b.purchase_date) AS recency,
            COUNT(*) AS frequency,
            COALESCE(SUM(b.amount), 0) AS monetary
        FROM base_purchase_data AS b
        GROUP BY b.user_id
    ),
    
    rfm_percent_rank AS (
        SELECT
            user_id,
            recency,
            frequency,
            monetary,
            -- DENSE_RANK를 이용해 동점자에게 완전히 동일한 순위를 부여한 후, 0~1 사이의 비율로 변환
            PERCENT_RANK() OVER (ORDER BY recency ASC) AS r_pct,
            PERCENT_RANK() OVER (ORDER BY frequency DESC) AS f_pct,
            PERCENT_RANK() OVER (ORDER BY monetary DESC) AS m_pct
        FROM user_rfm_raw
    )
    
    SELECT
        user_id,
        recency,
        frequency,
        monetary,
        -- 변환된 비율(0~1)을 기반으로 정확하게 5개 구간(Score 1~5)으로 매핑
        -- Recency: 작을수록 좋으므로 오름차순 정렬 기준, 상위 20%(비율 0.2 이하)가 5점
        CASE 
            WHEN r_pct <= 0.2 THEN 5
            WHEN r_pct <= 0.4 THEN 4
            WHEN r_pct <= 0.6 THEN 3
            WHEN r_pct <= 0.8 THEN 2
            ELSE 1 
        END AS r,
        -- Frequency & Monetary: 클수록 좋으므로 내림차순 정렬 기준, 상위 20%가 5점
        CASE 
            WHEN f_pct <= 0.2 THEN 5
            WHEN f_pct <= 0.4 THEN 4
            WHEN f_pct <= 0.6 THEN 3
            WHEN f_pct <= 0.8 THEN 2
            ELSE 1 
        END AS f,
        CASE 
            WHEN m_pct <= 0.2 THEN 5
            WHEN m_pct <= 0.4 THEN 4
            WHEN m_pct <= 0.6 THEN 3
            WHEN m_pct <= 0.8 THEN 2
            ELSE 1 
        END AS m
    FROM rfm_percent_rank
    ORDER BY m DESC, f DESC, r DESC;
  ```

  ```sql
            CASE 
                WHEN f_pct <= 0.2 THEN 5
                WHEN f_pct <= 0.4 THEN 4
                WHEN f_pct <= 0.6 THEN 3
                WHEN f_pct <= 0.8 THEN 2
                ELSE 1 
            END AS f,
            CASE 
                WHEN m_pct <= 0.2 THEN 5
                WHEN m_pct <= 0.4 THEN 4
                WHEN m_pct <= 0.6 THEN 3
                WHEN m_pct <= 0.8 THEN 2
                ELSE 1 
            END AS m
        FROM rfm_percent_rank
        ORDER BY m DESC, f DESC, r DESC)
    	
    ,
    mst_rfm_index as (
    select generate_series(1,5) as rfm_index),
    
    r_counts as (
    select
    	r as score,
    	count(*) as cnt
    from user_rfm_rank
    group by r),
    
    f_counts as (
    select
    	f as score,
    	count(*) as cnt
    from user_rfm_rank
    group by f),
    
    m_counts as (
    select
    	m as score,
    	count(*) as cnt
    from user_rfm_rank
    group by m)
    
    select
    m.rfm_index,
    coalesce(r.cnt,0) as r,
    coalesce(f.cnt,0) as f,
    coalesce(mc.cnt,0) as m
    from mst_rfm_index as m
    left join r_counts as r on m.rfm_index=r.score
    left join f_counts as f on m.rfm_index=f.score
    left join m_counts as mc on m.rfm_index=mc.score
    order by m.rfm_index desc
  ```

## 2차원 결합 분석
* R&F 매트릭스
  ```sql
    SELECT
    CONCAT('r_', r) AS r_rank,
    COUNT(CASE WHEN f = 5 THEN 1 END) AS f_5,
    COUNT(CASE WHEN f = 4 THEN 1 END) AS f_4,
    COUNT(CASE WHEN f = 3 THEN 1 END) AS f_3,
    COUNT(CASE WHEN f = 2 THEN 1 END) AS f_2,
    COUNT(CASE WHEN f = 1 THEN 1 END) AS f_1
    FROM user_rfm_rank
    GROUP BY r
    ORDER BY r DESC; -- 원본 숫자 r을 기준으로 역순 정렬 (r_5 -> r_1 순서)
  ```
* R-F 매트릭스에 M(평균 구매액) 결합하기
  ```sql
    SELECT
    CONCAT('r_', r) AS r_rank,
    -- 각 칸의 유저 수(Count)와 평균 구매액(Avg_M)을 동시에 확인
    COUNT(CASE WHEN f = 5 THEN 1 END) AS f_5_유저수,
    ROUND(AVG(CASE WHEN f = 5 THEN monetary END), 0) AS f_5_평균금액,
    
    COUNT(CASE WHEN f = 4 THEN 1 END) AS f_4_유저수,
    ROUND(AVG(CASE WHEN f = 4 THEN monetary END), 0) AS f_4_평균금액,
    
    COUNT(CASE WHEN f = 3 THEN 1 END) AS f_3_유저수,
    ROUND(AVG(CASE WHEN f = 3 THEN monetary END), 0) AS f_3_평균금액,
    
    COUNT(CASE WHEN f = 2 THEN 1 END) AS f_2_유저수,
    ROUND(AVG(CASE WHEN f = 2 THEN monetary END), 0) AS f_2_평균금액,
    
    COUNT(CASE WHEN f = 1 THEN 1 END) AS f_1_유저수,
    ROUND(AVG(CASE WHEN f = 1 THEN monetary END), 0) AS f_1_평균금액
    FROM user_rfm_rank
    GROUP BY r
    ORDER BY r DESC;
  ```
* 가치 중심의 F-M 매트릭스 분석 (Loyalty vs Value)
  ```sql
    SELECT
    CONCAT('f_', f) AS f_rank,
    COUNT(CASE WHEN m = 5 THEN 1 END) AS m_5_최고매출,
    COUNT(CASE WHEN m = 4 THEN 1 END) AS m_4_고매출,
    COUNT(CASE WHEN m = 3 THEN 1 END) AS m_3_중매출,
    COUNT(CASE WHEN m = 2 THEN 1 END) AS m_2_저매출,
    COUNT(CASE WHEN m = 1 THEN 1 END) AS m_1_최저매출
  ```

* 실무형 통합 고객 세그먼트
  ```sql
    WITH segmented_users AS (
        SELECT
            user_id,
            monetary,
            CASE
                WHEN r >= 4 AND f >= 4 AND m >= 4 THEN 'VIP 고객'
                WHEN r >= 4 AND f <= 2 THEN '신규 고객'
                WHEN r <= 2 AND f >= 4 THEN '이탈 위기 충성고객'
                WHEN r <= 2 AND f <= 2 THEN '휴면/이탈 고객'
                WHEN m >= 4 THEN '큰손 고객 (고단가)'
                ELSE '일반 고객'
            END AS user_segment
        FROM user_rfm_rank
    )

    SELECT
    user_segment,
    COUNT(user_id) AS 유저_수,
    ROUND(100.0 * COUNT(user_id) / SUM(COUNT(user_id)) OVER(), 1) AS 유저_비중_pct,
    SUM(monetary) AS 총_매출,
    ROUND(100.0 * SUM(monetary) / SUM(SUM(monetary)) OVER(), 1) AS 매출_비중_pct
    FROM segmented_users
    GROUP BY user_segment
    ORDER BY 총_매출 DESC;
  ```
