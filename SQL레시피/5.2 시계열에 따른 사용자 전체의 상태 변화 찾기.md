# 1. 등록 수의 추이와 경향 보기
## 월별 등록 수 추이 
```sql
with
monthly_agg as (
select
	to_char(register_date,'YYYY-MM') as year_month,
	count(distinct user_id) as register_count
from mst_users
group by 1)

select
year_month,
register_count,
lag(register_count) over (order by year_month) as last_month_count,
round(100.0*register_count / nullif(lag(register_count) over (order by year_month),0),2) as month_over_month_ratio
from monthly_agg
order by year_month
```

## 등록 디바이스별 추이
```sql
with
mst_users_with_year_month as (
select
	date_trunc('month',register_date)::date as year_month,
	user_id,
	register_device
from mst_users)

select
to_char(year_month,'YYYY-MM') as year_month,
count(user_id) as register_count,
COUNT(user_id) FILTER (WHERE register_device = 'PC_Web') AS register_pc,
COUNT(user_id) FILTER (WHERE register_device = 'Mobile_Android') AS register_android,
COUNT(user_id) FILTER (WHERE register_device = 'Mobile_iOS') AS register_ios
from mst_users_with_year_month
group by year_month
order by year_month
```

# 2.지속률과 정착률 산출하기
## 1~7일 정착률 계산하기 (실무용)
```sql
WITH target_users AS (
  SELECT
    user_id,
    register_date::date AS register_date,
    -- [안전장치] 데이터 분석 기준점을 '어제'로 고정
    (CURRENT_DATE - INTERVAL '1 day')::date AS yesterday_date
  FROM mst_users
  -- 오늘 가입한 유저는 어차피 어제 기준 데이터에 포함되지 않으므로 제외
  WHERE register_date::date < CURRENT_DATE
),

user_daily_activity AS (
  SELECT
    u.user_id,
    -- 가입일과 액션일의 날짜 차이 계산 (Day 1 ~ Day 7)
    (a.stamp::date - u.register_date) AS days_elapsed
  FROM target_users u
  INNER JOIN action_log a -- 가입 이후 액션이 있는 기록만 매칭
    ON u.user_id = a.user_id
   -- [성능 최적화] 타임스탬프 조건의 좌변 형변환(::date)을 제거하여 인덱스 스캔 유도
   -- 가입일 익일(Day 1)부터 7일차(Day 7) 범위의 로그만 스캔
   AND a.stamp >= u.register_date + INTERVAL '1 day'
   AND a.stamp < LEAST(CURRENT_DATE, u.register_date + INTERVAL '8 days')
  GROUP BY u.user_id, (a.stamp::date - u.register_date)
)

SELECT
  tu.register_date,
  -- [무결성 확보] LEFT JOIN을 통해 가입 후 아무 액션도 안 한 유저까지 포함한 실제 총 가입자 수 집계
  COUNT(DISTINCT tu.user_id) AS total_registrations,

  -- Day 1 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '1 day' <= tu.yesterday_date AND uda.days_elapsed = 1 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '1 day' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS day_1_rate,

  -- Day 2 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '2 day' <= tu.yesterday_date AND uda.days_elapsed = 2 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '2 day' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS day_2_rate,

  -- Day 3 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '3 day' <= tu.yesterday_date AND uda.days_elapsed = 3 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '3 day' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS day_3_rate,

  -- Day 4 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '4 day' <= tu.yesterday_date AND uda.days_elapsed = 4 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '4 day' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS day_4_rate,

  -- Day 5 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '5 day' <= tu.yesterday_date AND uda.days_elapsed = 5 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '5 day' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS day_5_rate,

  -- Day 6 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '6 day' <= tu.yesterday_date AND uda.days_elapsed = 6 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '6 day' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS day_6_rate,

  -- Day 7 정착률 (%)
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '7 day' <= tu.yesterday_date AND uda.days_elapsed = 7 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '7 day' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS day_7_rate

FROM target_users tu
LEFT JOIN user_daily_activity uda 
  ON tu.user_id = uda.user_id
GROUP BY tu.register_date
ORDER BY tu.register_date;
```
* **`total_registerations` 정확도 확보**
  
  * `target_users`를 메인(Driver) 테이블로 두고 액션 로그 유무와 상관없이 `left join`하여, 첫날 가입하고 바로 이탈한 유저까지 가입자 수(모수)에 온전히 집계되도록 수정.
* **대용량 로그 조회 최적화**
  
  * `user_daily_activity` 내의 `on`절에서 `a.stamp::date` 형변환 비교 대신 `a.stamp>=...`범위를 직접 비교함으로써 `action_log.stamp` 컬럼에 인덱스가 있을 경우 이를 100% 활용할 수 있도록 튜닝했다.
* **Interval 7 days** 최적화
  * 유저당 최대 가입 후 7일 동안의 로그만 탐색하므로, 테이블이 아무리 커도 조인 연산량이 대폭 제한된다.
 
* **시간적 기회 왜곡 차단**
  * 오늘이 6월 5일이라면 6월 4일 가입자의 Day 1은 6월 5일이다. 오늘이 아직 다 지나지 않았기 때문에 이 쿼리에서는 **거짓**이 되어 분모에서 안전하게 제외된다. --> 완벽하게 어제 하루를 통째로 보낸 유저들만 비교 대상이 되므로 데이터가 매우 정확해진다.
* `Least`
  * 나열된 여러 개의 값 중에서 최소값을 반환하는 역할.
    
  * `min()`: **세로(rows)** 를 비교하여 최소값을 찾는다.
    
  * `least()`:**가로(Arguments)** 를 비교해서 최소값을 찾는다.
    
  * `a.stamp < least(current_date, u.register_date _ interval '8 days'`: 로그를 최대 7일치까지만 읽되, 혹시 가입한 지 얼마 안 된 유저라면 오늘 날짜 전까지만 읽어라.
    * 시나리오 A: 가입한 지 오래된 유저 (예: 6월 1일 가입)
      * `u.register_date + interval '8 days' = 6월 9일
        
      * `current_date` = 6월 11일
        
      * `least(6/11,6/9)` = 6/9 선택
        
      * 결과: `a.stamp < '2026-06-09'` 가 되어 딱 7일치 로그만 깔끔하게 읽고 스캔을 종료
    * 시나리오 B: 가입한 지 2일 밖에 안 된 유저 (예: 6월 10일 가입)
      
      * `u.register_date + interval '8 days'` = 6월 18일 (아직 오지 않은 미래)
        
      * `current_date` = 6월 11일
        
      * `least(6/11,6/18)` = 6/11 선택
        
      * 결과: `a.stamp < '2026-06-11'`이 되어, 미래 데이터까지 범위를 잡지 않고 어제까지만 안전하게 로그를 읽는다.
* `tu.user_id`집계
  * `left join` 구조에서의 기준점을 유지
    
  * 분모(모수) 조건 제한에서의 정확성
    
    * `COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '1 day' <= tu.yesterday_date THEN tu.user_id END)`
      
      * 위 조건의 목적 = '가입한 지 1일 이상 지난 유저가 총 몇 명인가?'를 세는 것
        
      * 가입 후 아무것도 안 한 유저를 분모에 포함시켜야 정확한 리텐션이 나온다.
        
  * 분자(액션 유저) 조건에서의 작동 원리
    
    * `COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '1 day' <= tu.yesterday_date AND uda.days_elapsed = 1 THEN tu.user_id END)`
      
    * 가입 후 1일 차에 액션을 하지 않은 유저는 `uda.days_elased` 값이 `null` 이거나 다른 숫자일 것이기에, `case when` 조건 전체가 거짓(False 또는 Null)이 되어 결국 `tu.user_id`는 선택되지 않는다.

## Week 1 ~ Week 4 정착률 통합 쿼리 
```sql
WITH target_users AS (
  SELECT
    user_id,
    register_date::date AS register_date,
    -- [안전장치] 데이터 분석 기준점을 '어제'로 고정
    (CURRENT_DATE - INTERVAL '1 day')::date AS yesterday_date
  FROM mst_users
  -- 오늘 가입한 유저는 어차피 어제 기준 데이터에 포함되지 않으므로 제외
  WHERE register_date < CURRENT_DATE
),

user_weekly_activity AS (
  SELECT
    u.user_id,
    -- 가입일과 액션일의 날짜 차이 계산
    (a.stamp::date - u.register_date) AS days_elapsed
  FROM target_users u
  INNER JOIN action_log a 
    ON u.user_id = a.user_id
   -- [성능 최적화 1] 좌변 형변환(::date)을 제거하여 stamp 인덱스 스캔(Index Scan) 유도
   AND a.stamp >= u.register_date + INTERVAL '1 day'
   -- [성능 최적화 2] 4주차 최대 범위인 28일치 로그만 스캔하되, 오늘 날짜(CURRENT_DATE)를 넘지 않도록 방어
   AND a.stamp < LEAST(CURRENT_DATE, u.register_date + INTERVAL '29 days') 
  GROUP BY u.user_id, (a.stamp::date - u.register_date)
)

SELECT
  tu.register_date,
  -- [무결성 확보] LEFT JOIN을 통해 가입 후 단 한 번도 활동하지 않은 유저까지 온전히 집계
  COUNT(DISTINCT tu.user_id) AS total_registrations,

  -- Week 1 잔존율 (%) : 1~7일 사이 방문
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '7 days' <= tu.yesterday_date AND uda.days_elapsed BETWEEN 1 AND 7 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '7 days' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS retention_week_1,

  -- Week 2 잔존율 (%) : 8~14일 사이 방문
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '14 days' <= tu.yesterday_date AND uda.days_elapsed BETWEEN 8 AND 14 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '14 days' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS retention_week_2,

  -- Week 3 잔존율 (%) : 15~21일 사이 방문
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '21 days' <= tu.yesterday_date AND uda.days_elapsed BETWEEN 15 AND 21 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '21 days' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS retention_week_3,

  -- Week 4 잔존율 (%) : 22~28일 사이 방문
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '28 days' <= tu.yesterday_date AND uda.days_elapsed BETWEEN 22 AND 28 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '28 days' <= tu.yesterday_date THEN tu.user_id END), 0), 2) AS retention_week_4

FROM target_users tu
LEFT JOIN user_weekly_activity uda 
  ON tu.user_id = uda.user_id
GROUP BY tu.register_date
ORDER BY tu.register_date;
```

* **latest_date** --> **yesterday_date**로 변경 및 로직 단순화
  
  * 비용의 낭비 없이 **어제 날짜**를 기준점으로 삼는다.
    
* `a.stamp::date` < `current_date` 조건 추가
  
  * 오늘 쌓이고 있는 실시간 로그를 아예 읽지 않는다. --> 쿼리 스캔 비용을 획기적으로 줄인다.
    
* 완벽해진 정밀도
  
  * 어제 날짜를 기준으로 **register_date + N days <= yesterday_date** 만 분모에 들어가기 때문에, 5월 15일 가입자(Week3 종료일이 6월 5일 오늘인 유저)는 오늘이 다 지나지 않았으므로 안전하게 분모에서 제외된다. --> 이 유저는 내일이 되면 완벽한 데이터로 분모에 포함된다.
### 연습용 쿼리(snap shot 분석)

```sql
    (a.stamp::date - u.register_date) AS days_elapsed
  FROM target_users u
  INNER JOIN action_log a -- 활동 기록이 존재하는 유저 로그 매칭
    ON u.user_id = a.user_id
   -- [성능 최적화 2] 좌변의 형변환(::date)을 제거하여 stamp 컬럼의 인덱스 효율 극대화
   -- 가입일 익일(1일차)부터 최대 4주차(28일차) 범위의 로그만 스캔하도록 상한선 제약
   AND a.stamp >= u.register_date + INTERVAL '1 day'
   AND a.stamp < u.register_date + INTERVAL '29 days'
  GROUP BY u.user_id, (a.stamp::date - u.register_date)
)

SELECT
  tu.register_date,
  -- [무결성 확보] LEFT JOIN을 적용하여 4주간 활동이 전혀 없던 이탈 유저까지 실제 총 가입자 수에 포함
  COUNT(DISTINCT tu.user_id) AS total_registrations,

  -- Week 1 잔존율 (%) : 1~7일 사이 방문
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '7 days' <= tu.latest_date AND uda.days_elapsed BETWEEN 1 AND 7 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '7 days' <= tu.latest_date THEN tu.user_id END), 0), 2) AS retention_week_1,

  -- Week 2 잔존율 (%) : 8~14일 사이 방문
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '14 days' <= tu.latest_date AND uda.days_elapsed BETWEEN 8 AND 14 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '14 days' <= tu.latest_date THEN tu.user_id END), 0), 2) AS retention_week_2,

  -- Week 3 잔존율 (%) : 15~21일 사이 방문
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '21 days' <= tu.latest_date AND uda.days_elapsed BETWEEN 15 AND 21 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '21 days' <= tu.latest_date THEN tu.user_id END), 0), 2) AS retention_week_3,

  -- Week 4 잔존율 (%) : 22~28일 사이 방문
  ROUND(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '28 days' <= tu.latest_date AND uda.days_elapsed BETWEEN 22 AND 28 THEN tu.user_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN tu.register_date + INTERVAL '28 days' <= tu.latest_date THEN tu.user_id END), 0), 2) AS retention_week_4

FROM target_users tu
LEFT JOIN user_weekly_activity uda 
  ON tu.user_id = uda.user_id
GROUP BY tu.register_date
ORDER BY tu.register_date;
```
# 3. Aha! Moment /기능 채택과 리텐션의 상관관계
```sql
WITH target_users AS (
  SELECT
    user_id,
    register_date::date AS register_date,
    (CURRENT_DATE - INTERVAL '1 day')::date AS yesterday_date
  FROM mst_users
  -- 최소한 가입 익일(Day 1)이 어제 날짜보다 작거나 같은(이미 완벽히 지난) 유저만 선별
  WHERE register_date <= CURRENT_DATE - INTERVAL '2 days'
),

mst_actions AS (
  SELECT 'view' AS action
  UNION ALL SELECT 'comment' AS action
  UNION ALL SELECT 'follow' AS action
),

-- 1. 유저별 가입 당일(Day 0) 특정 액션 수행 여부 집계
user_day0_activity AS (
  SELECT
    u.user_id,
    a.action,
    1 AS do_action
  FROM target_users u
  INNER JOIN action_log a
    ON u.user_id = a.user_id
   -- [성능 최적화] 좌변 형변환을 제거하여 가입 당일(24시간) 로그만 인덱스 스캔
   AND a.stamp >= u.register_date
   AND a.stamp < u.register_date + INTERVAL '1 day'
  WHERE a.action IN ('view', 'comment', 'follow')
  GROUP BY u.user_id, a.action
),

-- 2. 유저별 가입 익일(Day 1) 재방문 여부 집계
user_day1_activity AS (
  SELECT
    u.user_id,
    1 AS index_date_action
  FROM target_users u
  INNER JOIN action_log a
    ON u.user_id = a.user_id
   -- [성능 최적화] 가입 익일(24~48시간 사이) 로그만 인덱스 스캔
   AND a.stamp >= u.register_date + INTERVAL '1 day'
   AND a.stamp < u.register_date + INTERVAL '2 days'
  GROUP BY u.user_id
)

-- 3. 최종 결합 및 액션별 리텐션 비교 집계
SELECT
  m.action,
  COUNT(DISTINCT u.user_id) AS users, -- 분석 대상이 되는 실제 총 유저 수
  
  -- 가입 당일 해당 액션 사용률 (%)
  ROUND(AVG(COALESCE(d0.do_action, 0)) * 100.0, 2) AS usage_rate,
  
  '01 day repeat' AS index_name,
  
  -- 가입 당일 액션을 '한' 유저들의 Day 1 잔존율 (%)
  ROUND(AVG(CASE WHEN COALESCE(d0.do_action, 0) = 1 THEN COALESCE(d1.index_date_action, 0) END) * 100.0, 2) AS idx_rate,
  
  -- 가입 당일 액션을 '안 한' 유저들의 Day 1 잔존율 (%)
  ROUND(AVG(CASE WHEN COALESCE(d0.do_action, 0) = 0 THEN COALESCE(d1.index_date_action, 0) END) * 100.0, 2) AS no_action_idx_rate

FROM target_users tu
CROSS JOIN mst_actions m 
LEFT JOIN user_day0_activity d0
  ON tu.user_id = d0.user_id
 AND m.action = d0.action
LEFT JOIN user_day1_activity d1
  ON tu.user_id = d1.user_id
GROUP BY m.action
ORDER BY m.action;
```
* 비교 집계 (A/B Test 형태)
  * 액션을 '한 유저'와 '안 한 유저'의 리텐션을 한눈에 대조해 보여준다.
    
  * **액션 아이템**을 도출하기에 완벽한 구조.

* `WHERE register_date <= CURRENT_DATE - INTERVAL '2 days'`: **Day1 재방문 여부(가입 후 2일차까지의 기록)가 완벽하게 다 쌓인 유저들만 분석 대상으로 삼겠다**는 데이터 분석의 **정밀 컷오프(Cut-off)** 안정장치
  
  * Day1 재방문까지 검증하려면 최소 이틀 전에 가입한 유저여야 데이터가 안전한데, 이를 가능하게 해준다.
